# TatvaOps Auth Service — Environment Variables

# ===========================================================
# SERVER
# ===========================================================
PORT=4500
NODE_ENV=production
SERVICE_NAME=tatvaops-auth-service

# Allowed origins (comma separated) for CORS
# Include frontend origins and other internal app origins
CORS_ORIGINS=https://app.primus9.ai,https://primus9.ai,https://www.primus9.ai

# Frontend app URL (where auth callbacks redirect back to)
FRONTEND_URL=https://app.primus9.ai

# ===========================================================
# DATABASE (shared PostgreSQL — same as backend)
# ===========================================================
DATABASE_URL=postgresql://user:password@localhost:5432/tatvaops_vision

# ===========================================================
# GOOGLE OAUTH 2.0
# ===========================================================
GOOGLE_CLIENT_ID=your_google_client_id_here
GOOGLE_CLIENT_SECRET=your_google_client_secret_here
# Must match exactly what you set in Google Cloud Console
GOOGLE_CALLBACK_URL=https://auth.primus9.ai/auth/google/callback

# ===========================================================
# JWT TOKENS
# ===========================================================
# Strong random secret — openssl rand -hex 64
JWT_SECRET=your_very_long_random_jwt_secret_here_minimum_64_chars

# Access token lifetime (e.g. 15m, 1h). Frontend refreshes before expiry; 1h = fewer refresh calls.
JWT_ACCESS_EXPIRY=1h

# Refresh token lifetime (e.g. 30d)
JWT_REFRESH_EXPIRY=30d

# Cookie domain (use .tatvaops.com in prod for cross-subdomain sharing)
COOKIE_DOMAIN=.primus9.ai
COOKIE_SECURE=true

# ===========================================================
# INTERNAL USER RULES
# ===========================================================
# Comma-separated email domains that are auto-marked as internal
INTERNAL_EMAIL_DOMAINS=tatvaops.com
